### Runnable focused regression check

Run from `interactive-explanation/`; uses installed Playwright and writes no test/baseline files:

```bash
node --input-type=module -e '
import assert from "node:assert/strict";
import { chromium } from "playwright";
import { createSmokeServer } from "./tools/smoke/server.mjs";
const server = await createSmokeServer({rootDir:process.cwd(),host:"127.0.0.1",port:4192,mountPath:"/interactive-explanation/"}).start();
const browser = await chromium.launch();
try {
  const page = await browser.newPage({viewport:{width:320,height:844}});
  await page.goto("http://127.0.0.1:4192/interactive-explanation/");
  await page.waitForFunction(() => document.querySelectorAll("[data-page-list] .page-card").length === 83);
  const query = page.locator("[data-filter-query]");
  await query.fill("game");
  await query.press("End");
  await query.press("Space");
  await query.press("t");
  assert.equal(await query.inputValue(), "game t");
  await page.locator("[data-clear-filters]").click();
  assert(await query.evaluate(e => e === document.activeElement));
  await page.locator(".advanced-filters summary").click();
  const family = page.locator("[data-atlas-family=teoria]");
  await family.focus();
  await page.keyboard.press("Space");
  assert(await family.evaluate(e => e === document.activeElement));
  assert.equal(await family.getAttribute("aria-pressed"), "true");
  await page.evaluate(() => document.documentElement.style.fontSize = "200%");
  assert(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth));
} finally {
  await browser.close();
  await new Promise(resolve => server.close(resolve));
}
'
```
